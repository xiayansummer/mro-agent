```python
# backend/app/platforms.py
"""平台注册表:平台 id / 中文名 / NL 别名 / 采集方式 的单一真源。

加新平台只在这里加一条数据;refine 的过滤指令、默认列表、扩展 vs 服务端路由、
前端文案(经 label map)全部从此派生,不再散落硬编码。
"""
import re

PLATFORM_REGISTRY: dict[str, dict] = {
    "jd":   {"cn": "京东工业品", "aliases": ["京东工业品", "京东", "jd"],            "collector": "extension"},
    "zkh":  {"cn": "震坤行",     "aliases": ["震坤行", "zkh"],                       "collector": "extension"},
    "ehsy": {"cn": "西域",       "aliases": ["西域", "ehsy"],                        "collector": "server"},
    "1688": {"cn": "阿里巴巴1688", "aliases": ["1688", "阿里巴巴", "阿里", "alibaba"], "collector": "extension"},
}

DEFAULT_PLATFORMS: list[str] = list(PLATFORM_REGISTRY)


def alias_to_id() -> dict[str, str]:
    return {alias: pid for pid, meta in PLATFORM_REGISTRY.items() for alias in meta["aliases"]}


def id_to_cn() -> dict[str, str]:
    return {pid: meta["cn"] for pid, meta in PLATFORM_REGISTRY.items()}


def aliases_by_length_desc() -> list[str]:
    aliases = [a for meta in PLATFORM_REGISTRY.values() for a in meta["aliases"]]
    return sorted(aliases, key=len, reverse=True)


def platform_alias_pattern() -> str:
    return "|".join(re.escape(a) for a in aliases_by_length_desc())


def cn_names() -> set[str]:
    return {meta["cn"] for meta in PLATFORM_REGISTRY.values()}


def extension_platform_ids() -> set[str]:
    return {pid for pid, meta in PLATFORM_REGISTRY.items() if meta["collector"] == "extension"}
```

```javascript
// extension/chrome/src/alibaba1688Parser.js
// ⚠️ 搜索结果页选择器为待校准初值(见 1688-calibrate.console.js);parsePrice 逻辑已完整可测。

const PRICE_RE = /([\d.]+)\s*[-~至到]\s*([\d.]+)|([\d.]+)/;

export function parsePrice(text) {
  const raw = (text || "").trim();
  const m = raw.match(PRICE_RE);
  if (!m) return { priceValue: null, priceText: raw };
  // 区间取下限;否则取单值。priceValue 是"起批价/区间下限",非真实批发底价。
  const low = m[1] != null ? parseFloat(m[1]) : parseFloat(m[3]);
  return { priceValue: Number.isFinite(low) ? low : null, priceText: raw };
}

function parseMoq(text) {
  const m = (text || "").match(/([\d]+)\s*(件|个|只|套|米|kg|千克)?起(批|订|定)?/);
  return m ? `≥${m[1]}${m[2] || "件"}` : "";
}

// ⚠️ 待真实页校准的选择器
const SEL = {
  card: ".offer-list-row-offer, [class*='offerCard'], .sm-offer-item",
  title: "[class*='title'], .title a",
  price: "[class*='price']",
  moq: "[class*='sale'], [class*='moq'], [class*='amount']",
  image: "img",
  link: "a[href*='detail'], a[href*='offer']",
};

export function parse1688SearchPage(limit = 10) {
  const offers = [];
  const cards = document.querySelectorAll(SEL.card);
  for (const card of cards) {
    if (offers.length >= limit) break;
    const title = (card.querySelector(SEL.title)?.textContent || "").trim();
    if (!title) continue;
    const { priceValue, priceText } = parsePrice(card.querySelector(SEL.price)?.textContent || "");
    const link = card.querySelector(SEL.link)?.href || "";
    const img = card.querySelector(SEL.image)?.src || "";
    offers.push({
      title: title.slice(0, 100),
      priceValue,
      priceText,
      moq: parseMoq(card.querySelector(SEL.moq)?.textContent || ""),
      imageUrl: img,
      productUrl: link,
      brand: null,
    });
  }
  const bodyText = document.body?.innerText || "";
  return {
    url: location.href,
    offers,
    hasLoginWall: /登录|signin|请登录/i.test(bodyText) && offers.length === 0,
    hasPriceSignal: offers.some((o) => o.priceValue != null),
  };
}

export function is1688SearchResultUrl(href) {
  return /1688\.com\/(s\/|offer_search|page\/offerlist)/i.test(href || "");
}

export function classify1688Page({ isSearchUrl, validOfferCount, hasLoginWall }) {
  if (hasLoginWall) return "login_required";
  if (isSearchUrl && validOfferCount > 0) return "ok";
  if (validOfferCount === 0) return "empty";
  return "ok";
}
```

```javascript
// extension/chrome/src/alibaba1688Search.js
import { parse1688SearchPage, is1688SearchResultUrl, classify1688Page } from "./alibaba1688Parser.js";
import { hasBrandMatch, normalizeRequiredBrand } from "./brandMatch.js";

const MAX_RESULTS_PER_TERM = 10;
const TERM_TIMEOUT_MS = 12000;

export async function run1688SearchTask(task) {
  const searchTerms = task.searchTerms || [];
  const requiredBrand = normalizeRequiredBrand(task.requiredBrand);
  let lastSearchTerm = "", lastError = "", bestPartial = null, loginRequired = false;

  for (const searchTerm of searchTerms) {
    lastSearchTerm = searchTerm;
    try {
      const page = await withTimeout(collect1688(searchTerm), TERM_TIMEOUT_MS, `1688 搜索超时：${searchTerm}`);
      const offers = page.offers || [];
      const verdict = classify1688Page({
        isSearchUrl: is1688SearchResultUrl(page.url),
        validOfferCount: offers.length,
        hasLoginWall: page.hasLoginWall,
      });
      if (verdict === "login_required") {
        loginRequired = true;
        lastError = "1688 登录态未知或被要求登录,未取到真实搜索结果";
        continue;
      }
      if (requiredBrand && offers.length > 0 && !hasBrandMatch(offers, requiredBrand)) {
        bestPartial ??= { searchTerm, offers };
        continue;
      }
      if (offers.length > 0) return { searchTerm, offers };
    } catch (error) {
      lastError = error.message || "1688 搜索失败";
    }
  }
  if (loginRequired && !bestPartial) return { loginRequired: true, searchTerm: lastSearchTerm, error: lastError };
  if (bestPartial) return bestPartial;
  return { searchTerm: lastSearchTerm, offers: [], error: lastError || "未解析到搜索结果" };
}

// (用 chrome.tabs + chrome.scripting.executeScript);实现细节参照 zkhSearch.js。
async function collect1688(term) {
  const url = `https://s.1688.com/selloffer/offer_search.htm?keywords=${encodeURIComponent(term)}`;
  // …仿 zkhSearch:打开/复用 tab、等待、executeScript 跑 parse1688SearchPage(MAX_RESULTS_PER_TERM)、返回其结果…
  // (参照 extension/chrome/src/zkhSearch.js 的 collectZkhSearchResults 实现)
  return await runInTab(url, () => parse1688SearchPage(MAX_RESULTS_PER_TERM));
}

function withTimeout(promise, ms, msg) {
  return Promise.race([promise, new Promise((_, rej) => setTimeout(() => rej(new Error(msg)), ms))]);
}
```
